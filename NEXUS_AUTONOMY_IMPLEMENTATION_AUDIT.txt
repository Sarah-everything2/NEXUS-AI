def initialize():
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)

    with sqlite3.connect(DB_PATH) as connection:
        connection.execute("""
            CREATE TABLE IF NOT EXISTS scheduled_tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                message TEXT NOT NULL,
                run_at TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'pending',
                created_at TEXT NOT NULL,
                repeat TEXT DEFAULT 'once'
            )
        """)

        columns = [
            row[1] for row in connection.execute(
                "PRAGMA table_info(scheduled_tasks)"
            )
        ]

        if "repeat" not in columns:
            connection.execute(
                "ALTER TABLE scheduled_tasks ADD COLUMN repeat TEXT DEFAULT 'once'"
            )

def add_reminder(minutes, message, repeat="once"):
    if minutes < 1:
        raise ValueError("Reminder must be at least 1 minute away.")

    if not message.strip():
        raise ValueError("Reminder message cannot be empty.")

    if repeat not in ("once", "hourly", "daily", "weekly"):
        raise ValueError("Invalid repeat interval.")

    initialize()

    run_at = datetime.now(timezone.utc) + timedelta(minutes=minutes)

    with sqlite3.connect(DB_PATH) as connection:
        cursor = connection.execute(
            """
            INSERT INTO scheduled_tasks
            (message, run_at, status, created_at, repeat)
            VALUES (?, ?, 'pending', ?, ?)
            """,
            (
                message.strip(),
                run_at.isoformat(),
                datetime.now(timezone.utc).isoformat(),
                repeat
            )
        )

        task_id = cursor.lastrowid

    schedule_job(task_id, message.strip(), run_at, repeat)

    return {
        "id": task_id,
        "message": message.strip(),
        "run_at": run_at.isoformat(),
        "repeat": repeat
    }

def schedule_job(task_id, message, run_at, repeat):
    if repeat == "once":
        scheduler.add_job(
            deliver_reminder,
            "date",
            run_date=run_at,
            args=[task_id, message],
            id=str(task_id),
            replace_existing=True
        )
    else:
        scheduler.add_job(
            deliver_reminder,
            "interval",
            **{
                "hours": 1 if repeat == "hourly" else 0,
                "days": 1 if repeat == "daily" else 0,
                "weeks": 1 if repeat == "weekly" else 0,
            },
            start_date=run_at,
            args=[task_id, message],
            id=str(task_id),
            replace_existing=True
        )

def cancel_reminder(task_id):
    initialize()

    with sqlite3.connect(DB_PATH) as connection:
        cursor = connection.execute("""
            UPDATE scheduled_tasks
            SET status = 'cancelled'
            WHERE id = ? AND status = 'pending'
        """, (task_id,))

    if cursor.rowcount:
        job = scheduler.get_job(str(task_id))
        if job:
            scheduler.remove_job(str(task_id))
        return True

    return False

def start_scheduler():
    initialize()

    if not scheduler.running:
        scheduler.start()

    schedule_mining_autopilot()
    schedule_video_autopilot()
    schedule_video_publisher()
    schedule_video_publication_monitor()

    with sqlite3.connect(DB_PATH) as connection:
        rows = connection.execute("""
            SELECT id, message, run_at, repeat
            FROM scheduled_tasks
            WHERE status = 'pending'
        """).fetchall()

    now = datetime.now(timezone.utc)

    for task_id, message, run_at, repeat in rows:
        scheduled_time = datetime.fromisoformat(run_at)

        if scheduled_time <= now:
            scheduled_time = now + timedelta(seconds=2)

        schedule_job(task_id, message, scheduled_time, repeat)

def save_tasks(tasks):
    if not isinstance(tasks, (list, tuple)):
        raise ValueError("Tasks must be a list.")

    data = _load()
    saved = []

    for task in tasks:
        if not isinstance(task, dict):
            continue

        task_id = _task_id(task)
        now = _now()
        existing = data["tasks"].get(task_id)

        if existing:
            existing["last_seen"] = now
            existing["discovery_count"] = existing.get("discovery_count", 1) + 1
            saved.append(existing)
            continue

        record = dict(task)
        record.update({
            "task_id": task_id,
            "status": "PREPARED",
            "created_at": now,
            "updated_at": now,
            "last_seen": now,
            "discovery_count": 1,
            "execution_started": False,
        })
        data["tasks"][task_id] = record
        saved.append(record)

    _save(data)
    return {"status": "SAVED", "tasks_saved": len(saved), "tasks": saved}

def get_tasks(status=None, limit=100):

def _call_adaptively(function, values):
    signature = inspect.signature(function)
    kwargs = {}

    for name, parameter in signature.parameters.items():
        if name in values:
            kwargs[name] = values[name]
        elif parameter.default is inspect.Parameter.empty:
            raise RuntimeError(
                f"Unsupported required engine parameter: {name}"
            )

    return function(**kwargs)

def _normalize_highlights(items):
    normalized = []

    if not isinstance(items, list):
        return normalized

    for item in items:
        if not isinstance(item, dict):
            continue

        try:
            start = float(item.get("start", item.get("start_seconds")))
            end = float(item.get("end", item.get("end_seconds")))

            if end <= start or start < 0:
                continue

            normalized.append({
                "start": start,
                "end": end,
                "duration": end - start,
                "score": float(item.get("score", item.get("rank", 0))),
                "text": str(item.get("text", item.get("reason", "")))[:1000]
            })
        except (TypeError, ValueError):
            continue

    return normalized

def process_job(job):
    job_id = job["id"]
    source = Path(job["source"]).resolve()

    if not source.is_file():
        raise FileNotFoundError("Source video no longer exists")

    digest = hashlib.sha256(
        str(source).encode("utf-8") +
        str(source.stat().st_size).encode("ascii")
    ).hexdigest()[:16]

    metadata_path = METADATA / f"{job_id}.json"
    transcript_path = TRANSCRIPTS / f"{job_id}.json"

    duplicate = video_enhancements.find_duplicate(source)
    if duplicate["duplicate"]:
        raise RuntimeError("Duplicate source video detected")

    # Probe source video.
    probe = engine.probe_video(str(source))

    if not isinstance(probe, dict):
        raise ValueError("Video probe returned invalid metadata")

    duration = float(
        probe.get("duration", 0) if probe else 0
    )

    if duration <= 0:
        raise ValueError("Video duration is missing or invalid")

    # Transcribe with existing Whisper implementation.
    transcription = _call_adaptively(
        engine.transcribe_video,
        {
            "video_path": str(source),
            "input_path": str(source),
            "model_size": "base",
            "model": "base",
            "language": None
        }
    )

    segments = _segments_from_transcription(transcription)

    transcript_path.write_text(
        json.dumps({
            "job_id": job_id,
            "source": str(source),
            "segments": segments,
            "created_at": time.time()
        }, indent=2, ensure_ascii=False),
        encoding="utf-8"
    )

    # Rank highlights using existing engine.
    raw_highlights = engine.detect_highlights(segments, top_k=10)
    highlights = _normalize_highlights(raw_highlights)

    outputs = []

    for index, highlight in enumerate(highlights, start=1):
        start = max(0, highlight["start"])
        end = min(duration, highlight["end"])
        clip_duration = end - start

        if clip_duration < 3:
            continue

        clip_name = f"{job_id}_short_{index:02d}.mp4"
        clip_path = SHORTS / clip_name

        engine.create_vertical_short(
            str(source),
            start,
            clip_duration,
            str(clip_path)
        )

        if not clip_path.is_file():
            continue

        caption_result = video_enhancements.create_captioned_clip(
            str(clip_path),
            segments,
            start,
            clip_duration,
            str(clip_path)
        )

        outputs.append({
            "video": caption_result.get("video", str(clip_path)),
            "subtitle": caption_result.get("subtitle"),
            "caption_status": caption_result.get("status"),
            "start": start,
            "duration": clip_duration,
            "score": highlight["score"],
            "text": highlight["text"],
            "platform_profiles": list(
                video_enhancements.PROFILES.keys()
            )
        })

    report = {
        "job_id": job_id,
        "source": str(source),
        "source_fingerprint": digest,
        "duration": duration,
        "transcript": str(transcript_path),
        "highlights_detected": len(highlights),
        "shorts_created": len(outputs),
        "outputs": outputs,
        "completed_at": time.time()
    }

    metadata_path.write_text(
        json.dumps(report, indent=2, ensure_ascii=False),
        encoding="utf-8"
    )

    try:
        report["manifest"] = video_enhancements.build_manifest(
            job_id, source, segments, outputs
        )
        video_enhancements.register_video(source, {
            "job_id": job_id,
            "duration": duration
        })
    except Exception as exc:
        report["manifest_error"] = type(exc).__name__

    return report

def update_job(job_id, status, error=None):

def recover_stale_jobs(max_age_seconds=3600):
    jobs = _read()
    recovered = 0
    now = time.time()

    for job in jobs:
        if job.get("status") != "PROCESSING":
            continue

        if now - float(job.get("updated_at", 0)) < max_age_seconds:
            continue

        job["status"] = (
            "QUEUED" if int(job.get("attempts", 0)) < 3 else "FAILED"

        recovered += 1

    if recovered:

def recover_interrupted():
    jobs = _read()
    count = 0

    for job in jobs:
        if job.get("status") != "PUBLISHING":
            continue

        # An interrupted network upload may already have reached
        # the platform. Never blindly retry an uncertain upload.
        job["status"] = "REVIEW_REQUIRED"
        job["last_error"] = (
            "Interrupted upload; reconcile remote platform state first"

def run_cycle(opportunities, electricity_price_per_kwh, hardware):
    """Evaluate mining. Execution occurs only when policy explicitly permits it."""
    policy = _read(POLICY)

    if EMERGENCY.exists():
        result = {"status": "EMERGENCY_STOP"}
        _audit(result)
        return result

    if policy.get("mining_enabled") is not True:
        return {"status": "DISABLED", "execution": False}

    decision = evaluate_opportunities(
        opportunities,
        electricity_price_per_kwh,
        hardware,
        settings=policy.get("profitability_limits", {})
    )

    if decision.get("status") != "RECOMMENDATION_READY":
        if policy.get("stop_when_unprofitable") is True:
            current = client.status()
            if current["status"] == "RUNNING":
                stopped = client.stop()
                result = {"status": "STOPPED_UNPROFITABLE", "stop": stopped,
                          "decision": decision}
                _audit(result)
                return result
        return {"status": decision.get("status"), "decision": decision,
                "execution": False}

    if policy.get("autonomous_execution_enabled") is not True:
        return {"status": "RECOMMENDATION_ONLY", "decision": decision,
                "execution": False}

    if EMERGENCY.exists():
        return {"status": "EMERGENCY_STOP", "execution": False}

    current = client.status()
    if current["status"] == "RUNNING":
        return {"status": "ALREADY_RUNNING", "decision": decision,
                "execution": False}

    result = client.start()
    output = {
        "status": result.get("status"),
        "decision": decision,
        "miner": result
    }
    _audit(output)
    return output

def bridge_listener_pid():
    """Return the PID listening on the bridge port, if any."""
    result = subprocess.run(
        ["powershell.exe", "-NoProfile", "-Command",
         "Get-NetTCPConnection -LocalPort 8001 -State Listen -ErrorAction SilentlyContinue | Select-Object -ExpandProperty OwningProcess -Unique"],
        capture_output=True, text=True, timeout=10,
    )
    if result.returncode != 0 or not result.stdout.strip():
        return None
    pids = {int(x) for x in result.stdout.split() if x.isdigit()}
    if len(pids) > 1:
        raise RuntimeError("Multiple processes reported on bridge port 8001.")
    return next(iter(pids), None)

async def _get(endpoint: str) -> dict:
    """Make a read-only request to the local Hermes bridge."""
    try:
        async with httpx.AsyncClient(
            base_url=BASE_URL,
            timeout=TIMEOUT,
            trust_env=False,
            follow_redirects=False,
        ) as client:
            response = await client.get(endpoint)
            response.raise_for_status()
            data = response.json()

        if not isinstance(data, dict):
            return {"status": "ERROR", "message": "Invalid bridge response."}

        if "error" in data:
            return {
                "status": "ERROR",
                "message": str(data["error"]),
            }

        return {
            "status": "SUCCESS",
            "data": data,
        }

    except httpx.TimeoutException:
        return {"status": "ERROR", "message": "MT5 bridge request timed out."}
    except httpx.HTTPStatusError as exc:
        return {
            "status": "ERROR",
            "message": f"MT5 bridge returned HTTP {exc.response.status_code}.",
        }
    except httpx.HTTPError:
        return {"status": "ERROR", "message": "MT5 bridge is unavailable."}
    except ValueError:
        return {"status": "ERROR", "message": "Invalid JSON from MT5 bridge."}

            const data = await response.json();

            if (!response.ok) {
